-- v1.0.15: REVIEW ON STAGING FIRST. THIS PROCEDURE DELETES WORKER DATA.
-- BACKUP before running. Do not execute against production until tested.
-- Uses SECURITY INVOKER; existing RLS on ktx_workers applies.
BEGIN;
CREATE OR REPLACE FUNCTION public.ktx_admin_delete_all_workers(p_expected_count bigint, p_phrase text)
RETURNS bigint LANGUAGE plpgsql SECURITY INVOKER SET search_path = public, pg_temp AS $$
DECLARE v_actual bigint; v_deleted bigint;
BEGIN
 IF auth.uid() IS NULL OR NOT EXISTS
  (SELECT 1 FROM public.app_accounts WHERE user_id=auth.uid() AND enabled=true AND role='admin')
 THEN RAISE EXCEPTION 'admin_required'; END IF;
 IF p_phrase IS DISTINCT FROM 'XÓA TẤT CẢ' THEN RAISE EXCEPTION 'invalid_confirmation'; END IF;
 IF p_expected_count IS NULL OR p_expected_count < 1 THEN RAISE EXCEPTION 'invalid_expected_count'; END IF;
 PERFORM pg_advisory_xact_lock(hashtext('ktx_delete_all_workers'));
 SELECT count(*) INTO v_actual FROM public.ktx_workers;
 IF v_actual <> p_expected_count THEN RAISE EXCEPTION 'count_mismatch: expected %, actual %',p_expected_count,v_actual; END IF;
 DELETE FROM public.ktx_workers;
 GET DIAGNOSTICS v_deleted = ROW_COUNT;
 IF v_deleted <> v_actual THEN RAISE EXCEPTION 'rls_delete_mismatch: expected %, deleted %',v_actual,v_deleted; END IF;
 RETURN v_deleted;
END $$;
REVOKE ALL ON FUNCTION public.ktx_admin_delete_all_workers(bigint,text) FROM PUBLIC, anon;
GRANT EXECUTE ON FUNCTION public.ktx_admin_delete_all_workers(bigint,text) TO authenticated;
COMMIT;
-- NOTE: No other tables are deleted. Foreign-key constraints may reject transaction safely.
